================================================================================
USAGE GUIDE FOR DEVELOPERS
================================================================================

Access the music controller from client code:

    local MusicClient = require(game.Players.LocalPlayer:WaitForChild("PlayerGui"):...)

    -- Play music for a game phase
    MusicClient.PlayTrackForPhase("BuildPhase")

    -- Play a specific track by name
    MusicClient.PlayTrack("Survival Mode")

    -- Control volume (0-1)
    MusicClient.SetMasterVolume(0.5)

    -- Mute/unmute
    MusicClient.SetMute(true)  -- Mute
    MusicClient.SetMute(false) -- Unmute

    -- Stop all music
    MusicClient.StopAllMusic()

    -- Get current track
    local trackName = MusicClient.GetCurrentTrackName()
